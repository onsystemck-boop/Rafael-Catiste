Codigo Apps Script:

function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  if (action === 'getData') {
    var config = getSheetData(ss, 'config');
    var imoveis = getSheetData(ss, 'imoveis');
    var depoimentos = getSheetData(ss, 'depoimentos');
    return jsonResponse({ config: config[0] || {}, imoveis: imoveis, depoimentos: depoimentos });
  }

  if (action === 'getLeads') {
    var leads = getSheetData(ss, 'leads');
    return jsonResponse({ leads: leads });
  }

  return jsonResponse({ error: 'action nao reconhecida' });
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var action = data.action;

  if (action === 'saveLead') {
    var sheet = getOrCreateSheet(ss, 'leads',
      ['Data','Nome','WhatsApp','Cidade','Idade','Estado Civil',
       'Renda','FGTS','Moradia','CLT/Servidor','Tipo Imovel','Urgencia','Obs']);
    sheet.appendRow([data.data, data.nome, data.whatsapp, data.cidade,
      data.idade, data.estadoCivil, data.renda, data.fgts,
      data.moradia, data.emprego, data.tipo, data.urgencia, data.obs]);
  }

  else if (action === 'saveImovel') {
    var im = data.imovel;
    var sheet = getOrCreateSheet(ss, 'imoveis',
      ['id','nome','local','preco','area','quartos','vagas','status','img','desc']);
    sheet.appendRow([im.id, im.nome, im.local, im.preco,
      im.area, im.quartos, im.vagas, im.status, im.img, im.desc]);
  }

  else if (action === 'deleteImovel') {
    var sheet = ss.getSheetByName('imoveis');
    if (sheet) {
      var vals = sheet.getDataRange().getValues();
      for (var i = vals.length - 1; i >= 1; i--) {
        if (String(vals[i][0]) === String(data.id)) {
          sheet.deleteRow(i + 1); break;
        }
      }
    }
  }

  else if (action === 'updateImovelStatus') {
    var sheet = ss.getSheetByName('imoveis');
    if (sheet) {
      var vals = sheet.getDataRange().getValues();
      for (var i = 1; i < vals.length; i++) {
        if (String(vals[i][0]) === String(data.id)) {
          sheet.getRange(i + 1, 8).setValue(data.status); break;
        }
      }
    }
  }

  else if (action === 'saveDepoimento') {
    var d = data.depoimento;
    var sheet = getOrCreateSheet(ss, 'depoimentos', ['nome','texto','cidade']);
    sheet.appendRow([d.nome, d.texto, d.cidade]);
  }

  else if (action === 'saveConfig') {
    var cfg = data.config;
    var sheet = getOrCreateSheet(ss, 'config', ['whatsapp','creci','adminPass']);
    sheet.clearContents();
    sheet.appendRow(['whatsapp','creci','adminPass']);
    sheet.appendRow([cfg.whatsapp || '', cfg.creci || '', cfg.adminPass || '']);
  }

  return jsonResponse({ ok: true });
}

function getSheetData(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet || sheet.getLastRow() < 2) return [];
  var vals = sheet.getDataRange().getValues();
  var headers = vals[0];
  var result = [];
  for (var i = 1; i < vals.length; i++) {
    var obj = {};
    headers.forEach(function(h, j) { obj[h] = vals[i][j]; });
    result.push(obj);
  }
  return result;
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  return sheet;
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
